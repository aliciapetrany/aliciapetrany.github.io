@njit(cache=True)
def _greedy_matching(ref_sorted, target_sorted, ws_sorted,
                     n_anchors_target, min_weight, n_cells):
    """
    This function facilitates anchor pairing

    Instead of using an MNN like selection approach, cells are greedily paired with their closest cross species
    relatives
    """
    claimed_ref = np.zeros(n_cells, dtype=np.bool_)
    claimed_target = np.zeros(n_cells, dtype=np.bool_)

    out_ref = np.empty(n_anchors_target, dtype=np.int32)
    out_target = np.empty(n_anchors_target, dtype=np.int32)
    out_w = np.empty(n_anchors_target, dtype=np.float32)
    n_found = 0

    for k in range(len(ref_sorted)):
        if ws_sorted[k] < min_weight:
            break
        ri = ref_sorted[k]
        ti = target_sorted[k]
        if claimed_ref[ri] or claimed_target[ti]:
            continue
        out_ref[n_found] = ri
        out_target[n_found] = ti
        out_w[n_found] = ws_sorted[k]
        claimed_ref[ri] = True
        claimed_target[ti] = True
        n_found += 1
        if n_found == n_anchors_target:
            break

    return out_ref[:n_found], out_target[:n_found], out_w[:n_found]

class CSHint_integration:
    """
    This module manages species integration using the merged matrix output by the CSHint_merge module.

    Steps:
      1) load() : Loads in the merged matrix
      2) select_features() : Selects a shortlist of features to use during integration
      3) prepare_for_integration(): Normalizes the data, runs PCA, and classifies cell types
      4) integrate() OR integrate_no_optimization() : Runs the TPS correction. integrate() is highly reccomended
      5) [ONLY IF NECESSARY] post_integration_correction() : removes cells that are not integrate properly
    """

    def load(self):
        """
        Re-load the merged matrix and its metadata from ``self.output_dir``.

        Reads ``matrix_normalised.mtx``, ``metadata.txt`` and ``rowmetadata.csv``, then
        computes per-cell-type means. Populates ``self.raw_tensor``,
        ``self.annotations``, ``self.raw_tensorlabels``,
        ``self.raw_gene_layer_assignments`` and ``self.cell_type_means``.

        :returns: ``None``. Results are stored on the instance.
        :rtype: None
        """
        print(f"Loading data from {self.output_dir}...")

        matrix_path = os.path.join(self.output_dir, "matrix_normalised.mtx")
        self.raw_tensor = sio.mmread(matrix_path).tocsr()

        col_meta_path = os.path.join(self.output_dir, "metadata.txt")
        col_df = pd.read_csv(col_meta_path, sep='\t')

        self.annotations = {}
        for sp in col_df['species'].unique():
            sp_df = col_df[col_df['species'] == sp]
            # Set index to barcode (removing species prefix for internal compatibility)
            self.annotations[sp] = pd.Series(
                sp_df['cell_type'].values,
                index=[bc.replace(f"{sp}_", "") for bc in sp_df['cell_id']]
            )

        self.raw_tensorlabels = {"cols": col_df['cell_id'].tolist()}

        row_meta_path = os.path.join(self.output_dir, "rowmetadata.csv")
        row_df = pd.read_csv(row_meta_path)
        self.raw_tensorlabels["rows"] = row_df['row_id'].tolist()

        self.raw_gene_layer_assignments = {}
        for _, row in row_df.iterrows():
            og = row['orthogroup']
            rank = int(row['rank'])
            genes = row['genes'].split(';')

            if og not in self.raw_gene_layer_assignments:
                self.raw_gene_layer_assignments[og] = {}
            for g in genes:
                self.raw_gene_layer_assignments[og][g] = rank

        self._calculate_means_from_matrix(col_df)



    def _calculate_means_from_matrix(self, col_df):
        """Computes means for each cell type per species from the sparse matrix."""
        print("Calculating cell type means from matrix...")
        self.cell_type_means = {}

        # Identify anchor cell types (present in all species)
        species_list = col_df['species'].unique()
        ct_per_sp = [set(col_df[col_df['species'] == s]['cell_type']) for s in species_list]
        anchor_cts = sorted(list(set.intersection(*ct_per_sp)))

        #get species-wise cell type anchors across species
        for sp in species_list:
            sp_mask = (col_df['species'] == sp).values
            sp_matrix = self.raw_tensor[:, sp_mask]  # Matrix is (Genes, Cells)
            sp_metadata = col_df[sp_mask]

            means_dict = {}
            for ct in anchor_cts:
                ct_mask = (sp_metadata['cell_type'] == ct).values
                if ct_mask.sum() > 0:
                    # Mean across columns (cells) for this cell type
                    means_dict[ct] = np.asarray(sp_matrix[:, ct_mask].mean(axis=1)).flatten()
                else:
                    means_dict[ct] = np.zeros(sp_matrix.shape[0])

            # Index by row_id (the merged features)
            self.cell_type_means[sp] = pd.DataFrame(means_dict, index=self.raw_tensorlabels["rows"]).T

    def select_features(self, discriminability_threshold=0.75, conservation_threshold=0.5):
        """
        Select features that are both conserved across species and discriminative
        between cell types.

        Conservation is scored first and filters the candidate pool; the discriminability
        cutoff is then a percentile of the survivors only, so the two thresholds are not
        independent. Populates ``self.selected_features``,
        ``self.discriminability_scores`` and ``self.conservation_scores``.

        :param discriminability_threshold: Percentile cutoff on discriminability,
            expressed as a fraction and applied to the conservation-filtered candidates.
            This is a quantile, not a score: ``0.75`` keeps the most variable quarter of
            candidates, and ``0.0`` keeps all of them.
        :type discriminability_threshold: float
        :param conservation_threshold: Minimum mean pairwise Spearman correlation across
            species for a feature to be considered at all. Applied as an absolute score
            in ``[-1, 1]``, unlike the discriminability cutoff.
        :type conservation_threshold: float
        :returns: ``None``. Results are stored on the instance.
        :rtype: None
        :raises IndexError: If no feature clears ``conservation_threshold``, leaving the
            percentile with an empty distribution.
        """
        species_list = list(self.cell_type_means.keys())
        discriminability_threshold = discriminability_threshold * 100
        # Get unique features from the row metadata
        unique_features = self.raw_tensorlabels["rows"]

        discriminability_scores = {}
        conservation_scores = {}
        final_candidates = []

        for feat in unique_features:
            vecs = []
            for species in species_list:
                # Direct lookup since index is now the feature name
                v = self.cell_type_means[species][feat].values
                vecs.append(v)

            # Discriminability (Variance)
            discriminability_scores[feat] = np.mean([np.var(v) for v in vecs])

            # Conservation (Correlation)
            corrs = []
            for i in range(len(vecs)):
                for j in range(i + 1, len(vecs)):
                    c, _ = spearmanr(vecs[i], vecs[j])
                    if not np.isnan(c): corrs.append(c)

            conservation_scores[feat] = np.mean(corrs) if corrs else 0

            if conservation_scores[feat] >= conservation_threshold:
                final_candidates.append(feat)

        # Apply Percentile Threshold
        disc_values = [discriminability_scores[f] for f in final_candidates]
        disc_cutoff = np.percentile(disc_values, discriminability_threshold)

        # Store as (og, rank) tuples to keep compatibility with rest of class
        self.selected_features = []
        for f in final_candidates:
            if discriminability_scores[f] >= disc_cutoff:
                # Reconstruct the tuple from the string name "OG_rankX"
                parts = f.split('_rank')
                self.selected_features.append((parts[0], int(parts[1])))

        self.discriminability_scores = {(f.split('_rank')[0], int(f.split('_rank')[1])): v
                                        for f, v in discriminability_scores.items()}
        self.conservation_scores = {(f.split('_rank')[0], int(f.split('_rank')[1])): v
                                    for f, v in conservation_scores.items()}

        print(f"Selected {len(self.selected_features)} features.")

    def to_anndata(self):
        """Convert the feature-by-cell matrix into a cell-by-feature AnnData.

        Slices ``self.raw_tensor`` to the selected features and transposes it, attaching
        orthogroup, rank and score columns to ``var`` and species and cell type to
        ``obs``.

        :returns: Cell-by-feature data for the selected features.
        :rtype: anndata.AnnData
        :raises ValueError: If :meth:`select_features` has not been run, or if no
            selected feature name matches a row of the matrix.
        """
        if not hasattr(self, "selected_features"):
            raise ValueError("Run select_features() first.")

        # 1. Map feature names to matrix row indices
        row_names = self.raw_tensorlabels["rows"]
        row_to_idx = {name: i for i, name in enumerate(row_names)}

        indices = []
        valid_features = []
        for og_name, rank in self.selected_features:
            fname = f"{og_name}_rank{rank}"

            if fname in row_to_idx:
                indices.append(row_to_idx[fname])
                valid_features.append((og_name, rank, fname))

        if not indices:
            raise ValueError(
                f"No features found in matrix! Check if row names in your "
                f"matrix (e.g., '{row_names[0]}') match expected format '{og_name}_rank{rank}'"
            )

        X_selected = self.raw_tensor[indices, :].T.tocsr()

        var_df = pd.DataFrame({
            "orthogroup": [f[0] for f in valid_features],
            "rank": [f[1] for f in valid_features],  # Changed 'layer' to 'rank' for consistency
            "discriminability": [self.discriminability_scores[(f[0], f[1])] for f in valid_features],
            "conservation": [self.conservation_scores[(f[0], f[1])] for f in valid_features],
        }, index=[f[2] for f in valid_features])

        barcodes = self.raw_tensorlabels["cols"]

        species_map = {}
        type_map = {}
        for species, ann in self.annotations.items():
            for bc, val in ann.items():
                full_bc = f"{species}_{bc}"
                species_map[full_bc] = species
                type_map[full_bc] = val

        obs_df = pd.DataFrame({
            "species": [species_map.get(bc, "unknown") for bc in barcodes],
            "cell_type": [type_map.get(bc, "unknown") for bc in barcodes]
        }, index=barcodes)

        adata = ad.AnnData(X=X_selected, obs=obs_df, var=var_df)
        return adata

    def prepare_for_integration(self, n_components=50):
        """Build the AnnData object and PCA embedding consumed by :meth:`optimize`.

        Runs total-count normalisation to 10,000 counts per cell, ``log1p``, unit scaling
        clipped at 10 standard deviations, then PCA. Also classifies cell types, which
        populates ``self.pair_anchors`` as a side effect on the instance even though the
        returned object is not stored.

        :param n_components: Number of principal components to compute. Must not exceed
            ``min(n_cells, n_features)``.
        :type n_components: int
        :returns: Preprocessed data with the embedding in ``obsm["X_pca"]``.
        :rtype: anndata.AnnData
        """
        adata = self.to_anndata()
        sc.pp.normalize_total(adata, target_sum=1e4)
        sc.pp.log1p(adata)
        sc.pp.scale(adata, max_value=10)
        sc.tl.pca(adata, n_comps=n_components)
        self._classify_cell_types()

        return adata



    def _classify_cell_types(self):
        species_list = list(self.annotations.keys())
        n_species = len(species_list)

        all_cell_types = set()
        for ann in self.annotations.values():
            all_cell_types.update(ann.unique())

        self.cell_type_species = {
            ct: [s for s in species_list if ct in self.annotations[s].unique()]
            for ct in all_cell_types
        }

        # Global anchor classifications (kept for reference / reporting)
        self.full_anchors = [
            ct for ct, sps in self.cell_type_species.items() if len(sps) == n_species
        ]
        self.partial_anchors = [
            ct for ct, sps in self.cell_type_species.items() if 1 < len(sps) < n_species
        ]
        self.unique_cell_types = [
            ct for ct, sps in self.cell_type_species.items() if len(sps) == 1
        ]

        if self.reference_species is None:
            self.reference_species = max(
                species_list, key=lambda s: len(self.annotations[s])
            )

        # Per-pair anchors: for each target species, cell types shared with reference
        ref = self.reference_species
        ref_cell_types = set(self.annotations[ref].unique())
        self.pair_anchors = {}
        for s in species_list:
            if s == ref:
                continue
            target_cell_types = set(self.annotations[s].unique())
            shared = sorted(ref_cell_types & target_cell_types)
            self.pair_anchors[s] = shared

        print(f"  Reference species: {self.reference_species} "
              f"({self.annotations[self.reference_species].nunique()} cell types)")
        print(f"  Full anchors (all species): {len(self.full_anchors)}")
        print(f"  Partial anchors: {len(self.partial_anchors)}")
        print(f"  Unique cell types: {len(self.unique_cell_types)}")
        for s, anchors in self.pair_anchors.items():
            print(f"  {ref} <-> {s}: {len(anchors)} shared cell types "
                  f"({', '.join(anchors[:5])}{'...' if len(anchors) > 5 else ''})")

    def _precompute_anchor_candidates(self, adata, X_ranked=None,
                                      cell_type_agnostic=False, chunk_size=512,
                                      min_anchor_weight=0.3):
        """Compute and cache ranked anchor candidate pairs for every non-reference
        species. Populates ``self.anchor_candidates``; read-only thereafter."""
        print("  Precomputing anchor candidates...")

        X_pca = adata.obsm["X_pca"]
        species_labels = adata.obs["species"].values
        cell_type_labels = adata.obs["cell_type"].values
        ref = self.reference_species
        species_list = list(self.annotations.keys())

        if X_ranked is None:
            X_ranked = np.apply_along_axis(rankdata, 0, X_pca).astype(np.float32)
            X_ranked -= X_ranked.mean(axis=1, keepdims=True)
            norms = np.linalg.norm(X_ranked, axis=1, keepdims=True)
            norms[norms == 0] = 1.0
            X_ranked /= norms

        self.anchor_candidates = {}
        self.anchor_cell_types = cell_type_labels
        self.anchor_species_labels = species_labels
        self.min_anchor_weight = min_anchor_weight

        for s_target in species_list:
            if s_target == ref:
                continue

            all_ref = []
            all_target = []
            all_w = []

            if cell_type_agnostic:
                ref_idx = np.where(species_labels == ref)[0]
                target_idx = np.where(species_labels == s_target)[0]
                if len(ref_idx) == 0 or len(target_idx) == 0:
                    continue

                X_ref = X_ranked[ref_idx]
                X_target = X_ranked[target_idx]
                ws = np.empty((len(ref_idx), len(target_idx)), dtype=np.float32)
                for start in range(0, len(ref_idx), chunk_size):
                    end = min(start + chunk_size, len(ref_idx))
                    ws[start:end] = X_ref[start:end] @ X_target.T

                all_ref.append(np.repeat(ref_idx, len(target_idx)))
                all_target.append(np.tile(target_idx, len(ref_idx)))
                all_w.append(ws.ravel())

            else:
                pair_shared = self.pair_anchors.get(s_target, [])
                if not pair_shared:
                    print(f"  Skipping {s_target} — no shared cell types with {ref}")
                    continue

                print(f"  Running cell-type-aware anchor selection: "
                      f"{ref} <-> {s_target} ({len(pair_shared)} shared types)")

                for cell_type in pair_shared:
                    ref_ct_idx = np.where(
                        (species_labels == ref) &
                        (cell_type_labels == cell_type)
                    )[0]
                    target_ct_idx = np.where(
                        (species_labels == s_target) &
                        (cell_type_labels == cell_type)
                    )[0]

                    if len(ref_ct_idx) == 0 or len(target_ct_idx) == 0:
                        continue

                    X_ref = X_ranked[ref_ct_idx]
                    X_target = X_ranked[target_ct_idx]
                    ws = np.empty(
                        (len(ref_ct_idx), len(target_ct_idx)), dtype=np.float32
                    )
                    for start in range(0, len(ref_ct_idx), chunk_size):
                        end = min(start + chunk_size, len(ref_ct_idx))
                        ws[start:end] = X_ref[start:end] @ X_target.T

                    all_ref.append(np.repeat(ref_ct_idx, len(target_ct_idx)))
                    all_target.append(np.tile(target_ct_idx, len(ref_ct_idx)))
                    all_w.append(ws.ravel())

            if not all_ref:
                print(f"  Skipping {s_target} — no eligible cell type overlaps")
                continue

            ref_flat = np.concatenate(all_ref).astype(np.int32)
            target_flat = np.concatenate(all_target).astype(np.int32)
            ws_flat = np.concatenate(all_w).astype(np.float32)

            keep = ws_flat >= min_anchor_weight
            ref_flat = ref_flat[keep]
            target_flat = target_flat[keep]
            ws_flat = ws_flat[keep]

            if len(ws_flat) == 0:
                print(f"  Skipping {s_target} — no pairs above min_anchor_weight")
                continue

            order = np.argsort(-ws_flat)
            self.anchor_candidates[s_target] = (
                ref_flat[order],
                target_flat[order],
                ws_flat[order],
            )

            n_shared = (len(self.pair_anchors[s_target])
                        if not cell_type_agnostic else 0)
            print(f"  {ref} x {s_target}: {len(order)} candidates across "
                  f"{n_shared} shared cell types | "
                  f"weight range: [{ws_flat.min():.3f}, {ws_flat.max():.3f}]")

        print("  Anchor candidates precomputed.")

    def _build_anchors(self, n_anchors_target=500, cell_type_agnostic=False,
                       min_anchor_weight=None):
        """Greedily match cached candidates into a mutually exclusive anchor set.

        Returns the anchor list rather than mutating ``self`` so that concurrent
        Optuna trials cannot interfere with one another."""
        if not hasattr(self, "anchor_candidates"):
            raise ValueError(
                "Call _precompute_anchor_candidates() before _build_anchors()."
            )

        cell_type_labels = self.anchor_cell_types
        n_cells = len(self.anchor_species_labels)
        min_w = float(min_anchor_weight) if min_anchor_weight is not None else -np.inf
        anchors_all = []

        for s_target, (ref_sorted, target_sorted, ws_sorted) in \
                self.anchor_candidates.items():

            matched_ref, matched_target, matched_w = _greedy_matching(
                ref_sorted, target_sorted, ws_sorted,
                n_anchors_target, min_w, n_cells,
            )

            n_found = len(matched_ref)
            if n_found < n_anchors_target:
                print(f"  {s_target}: only {n_found} anchors available "
                      f"(requested {n_anchors_target})")

            anchors = [
                {
                    "reference_cell": int(ri),
                    "target_cell": int(ti),
                    "target_species": s_target,
                    "cell_type": cell_type_labels[ri] if not cell_type_agnostic else None,
                    "weight": float(w),
                }
                for ri, ti, w in zip(matched_ref, matched_target, matched_w)
            ]

            anchors_all.extend(anchors)
            if anchors:
                print(f"  {s_target}: {len(anchors)} anchors | "
                      f"mean weight: {matched_w.mean():.3f}")

        print(f"  {len(anchors_all)} total anchors "
              f"across {len(self.anchor_candidates)} species pairs")
        return anchors_all

    def _fit_tps_correction(self, adata, anchors, regularisation=1.0):
        """Fit and apply a per-species thin-plate-spline warp onto the reference
        embedding, using ``anchors`` as control points. Mutates ``adata.obsm``."""
        print(f"  Fitting TPS correction ({len(anchors)} control points)...")

        X_pca = adata.obsm["X_pca"].copy()
        n_cells, n_components = X_pca.shape
        species_labels = adata.obs["species"].values
        ref = self.reference_species
        X_pca_corrected = X_pca.copy()

        def tps_kernel(D_sq):
            with np.errstate(divide="ignore", invalid="ignore"):
                return np.where(D_sq == 0, 0.0, D_sq * np.log(np.sqrt(D_sq) + 1e-10))

        def sq_distance_matrix(A, B):
            A_sq = (A ** 2).sum(axis=1, keepdims=True)
            B_sq = (B ** 2).sum(axis=1, keepdims=True)
            return np.maximum(A_sq + B_sq.T - 2 * (A @ B.T), 0.0)

        for target_species in self.annotations.keys():
            if target_species == ref:
                continue

            species_anchors = [
                a for a in anchors
                if a["target_species"] == target_species
            ]
            if not species_anchors:
                print(f"  No anchors for {target_species}, skipping")
                continue

            print(f"  Fitting TPS for {target_species} "
                  f"({len(species_anchors)} control points)...")

            source_points = np.array(
                [X_pca[a["target_cell"]] for a in species_anchors],
                dtype=np.float32
            )
            target_points = np.array(
                [X_pca[a["reference_cell"]] for a in species_anchors],
                dtype=np.float32
            )

            n_anchors = len(species_anchors)

            D_sq = sq_distance_matrix(source_points, source_points)
            K = tps_kernel(D_sq) + regularisation * np.eye(n_anchors, dtype=np.float32)
            P = np.hstack([
                np.ones((n_anchors, 1), dtype=np.float32), source_points
            ])
            n_affine = n_components + 1

            system = np.vstack([
                np.hstack([K, P]),
                np.hstack([P.T, np.zeros((n_affine, n_affine), dtype=np.float32)])
            ])
            rhs = np.vstack([
                target_points,
                np.zeros((n_affine, n_components), dtype=np.float32)
            ])

            try:
                coeffs = np.linalg.solve(system, rhs)
            except np.linalg.LinAlgError:
                print(f"    Warning: TPS solve failed for {target_species}, "
                      f"falling back to least squares")
                coeffs, _, _, _ = np.linalg.lstsq(system, rhs, rcond=None)

            w = coeffs[:n_anchors]
            a = coeffs[n_anchors:]

            species_mask = species_labels == target_species
            X_species = X_pca[species_mask].astype(np.float32)
            n_species_cells = X_species.shape[0]
            chunk_size = 1024

            X_corrected_species = np.empty_like(X_species)
            for start in range(0, n_species_cells, chunk_size):
                end = min(start + chunk_size, n_species_cells)
                X_chunk = X_species[start:end]
                D_sq_chunk = sq_distance_matrix(X_chunk, source_points)
                K_chunk = tps_kernel(D_sq_chunk)
                P_chunk = np.hstack([
                    np.ones((end - start, 1), dtype=np.float32), X_chunk
                ])
                X_corrected_species[start:end] = K_chunk @ w + P_chunk @ a

            X_pca_corrected[species_mask] = X_corrected_species

            anchor_source_globals = np.array(
                [a["target_cell"] for a in species_anchors]
            )
            species_global_indices = np.where(species_mask)[0]
            local_positions = np.searchsorted(species_global_indices,
                                              anchor_source_globals)
            corrected_anchors = X_corrected_species[local_positions]
            mean_dist = np.sqrt(
                ((corrected_anchors - target_points) ** 2).sum(axis=1)
            ).mean()
            print(f"    Mean post-correction distance to target: {mean_dist:.4f}")

        adata.obsm["X_pca_tps"] = X_pca_corrected
        adata.obsm["X_pca"] = X_pca_corrected
        print("  TPS correction complete")
        return adata

    def integrate(
            self,
            adata_pca,
            n_trials=10,
            n_jobs=1,
            random_state=42,
            metric_weights=None,
            regularisation=1.0,
            cell_type_agnostic=False,
            min_anchor_weight=0.3,
    ):
        """Optimise ``n_anchors_target`` for TPS-based cross-species integration.

        Anchor candidate correlations are computed once up front and shared read-only
        across trials; each trial then runs only the greedy match, the TPS warp and the
        scib metrics. Anchors pass between those stages by value, so trials stay
        independent when ``n_jobs > 1``. Once the study completes, anchors are rebuilt
        with the winning parameters and the correction is applied to ``adata_pca`` in
        place, leaving ``self.anchors`` and ``self.best_params`` in agreement with it.

        Scored metrics are ARI, NMI and graph connectivity;

        :param adata_pca: Data carrying an uncorrected ``obsm["X_pca"]`` plus
            ``obs["species"]`` and ``obs["cell_type"]``. Modified in place once the study
            has finished.
        :type adata_pca: anndata.AnnData
        :param n_trials: Number of Optuna trials to run.
        :type n_trials: int
        :param n_jobs: Number of trials to run concurrently.
        :type n_jobs: int
        :param random_state: Seed for the TPE sampler, the neighbour graph and Leiden.
        :type random_state: int
        :param metric_weights: Weights keyed by ``"ari"``, ``"nmi"`` and
            ``"graph_connectivity"``, normalised to sum to one. Defaults to
            ``{"ari": 0.4, "nmi": 0.2, "graph_connectivity": 0.4}``.
        :type metric_weights: dict[str, float] or None
        :param regularisation: Ridge term on the TPS kernel diagonal. Fixed across trials
            rather than optimised.
        :type regularisation: float
        :param cell_type_agnostic: If True, match anchors across the whole species pair
            rather than within shared cell types.
        :type cell_type_agnostic: bool
        :param min_anchor_weight: Minimum rank correlation for a candidate pair to be
            retained during precomputation.
        :type min_anchor_weight: float
        :returns: The corrected anndata object, the completed study and its best parameter dictionary.
        :rtype: tuple[adata, ptuna.study.Study, dict]
        :raises ValueError: If :meth:`_classify_cell_types` has not been run, or if fewer
            than two shared cell types exist across all species pairs.
        """
        import scib

        if not hasattr(self, "pair_anchors"):
            raise ValueError("_classify_cell_types() must be run before optimize().")

        if metric_weights is None:
            metric_weights = {
                "ari": 0.4,
                "nmi": 0.2,
                "graph_connectivity": 0.4,
            }
        total_w = sum(metric_weights.values())
        metric_weights = {k: v / total_w for k, v in metric_weights.items()}

        # Evaluate against the union of all per-pair shared cell types.
        all_pair_anchor_types = set()
        for shared in self.pair_anchors.values():
            all_pair_anchor_types.update(shared)

        if len(all_pair_anchor_types) < 2:
            raise ValueError(
                f"Need at least 2 shared cell types across pairs, "
                f"found {len(all_pair_anchor_types)}."
            )

        cell_type_labels = adata_pca.obs["cell_type"].values
        anchor_mask = np.array([ct in all_pair_anchor_types for ct in cell_type_labels])
        true_labels_anchor = cell_type_labels[anchor_mask]

        self._precompute_anchor_candidates(
            adata_pca,
            cell_type_agnostic=cell_type_agnostic,
            min_anchor_weight=min_anchor_weight,
        )

        # Smallest unique-cell count across all pairs bounds the search.
        max_anchors = min(
            min(
                len(np.unique(ref_sorted)),
                len(np.unique(target_sorted))
            )
            for ref_sorted, target_sorted, _ in self.anchor_candidates.values()
        )

        print(
            f"  Optimizing over {n_trials} trials\n"
            f"  Parameter: n_anchors_target in [10, {max_anchors}]\n"
            f"  Regularisation fixed at {regularisation}\n"
            f"  Metric weights: {metric_weights}"
        )

        def objective(trial):
            n_anchors_target = trial.suggest_int("n_anchors_target", 10, max_anchors)
            adata_work = adata_pca.copy()
            try:
                anchors = self._build_anchors(
                    n_anchors_target=n_anchors_target,
                    cell_type_agnostic=cell_type_agnostic,
                )

                if not anchors:
                    raise optuna.exceptions.TrialPruned("No anchors selected.")

                adata_work = self._fit_tps_correction(
                    adata_work, anchors, regularisation=regularisation
                )

                sc.pp.neighbors(adata_work, use_rep="X_pca", random_state=random_state)
                sc.tl.leiden(adata_work, random_state=random_state)

                adata_work.obs["cell_type"] = adata_work.obs["cell_type"].astype("category")
                adata_work.obs["species"] = adata_work.obs["species"].astype("category")
                adata_work.obs["leiden"] = adata_work.obs["leiden"].astype("category")

                cluster_labels = adata_work.obs["leiden"].values
                predicted_anchor = cluster_labels[anchor_mask]

                scores = {
                    "ari": adjusted_rand_score(
                        true_labels_anchor, predicted_anchor
                    ),
                    "nmi": normalized_mutual_info_score(
                        true_labels_anchor, predicted_anchor,
                        average_method="arithmetic",
                    ),
                    "graph_connectivity": scib.me.graph_connectivity(
                        adata_work, label_key="cell_type"
                    ),
                }

                weighted_score = sum(
                    metric_weights.get(k, 0.0) * v for k, v in scores.items()
                )
                for k, v in scores.items():
                    trial.set_user_attr(k, float(v))
                trial.set_user_attr("weighted_score", float(weighted_score))
                trial.set_user_attr("n_anchors", len(anchors))

            except optuna.exceptions.TrialPruned:
                raise
            except Exception as e:
                import traceback
                traceback.print_exc()
                raise optuna.exceptions.TrialPruned(f"Trial failed: {e}")

            return weighted_score

        sampler = optuna.samplers.TPESampler(seed=random_state)
        study = optuna.create_study(direction="maximize", sampler=sampler)
        study.optimize(
            objective, n_trials=n_trials, n_jobs=n_jobs, show_progress_bar=True
        )

        self.best_params = study.best_params

        best = study.best_trial
        print(f"\n  Best weighted score: {study.best_value:.4f}")
        print(f"  Best params:         {study.best_params}")
        print(f"  n_anchors:           {best.user_attrs.get('n_anchors')}")
        print(f"  Individual scores:   "
              f"{ {k: round(best.user_attrs.get(k, 0), 4) for k in metric_weights} }")

        # Refit on the winning parameters
        print("\n  Refitting with best parameters...")
        self.anchors = self._build_anchors(
            n_anchors_target=study.best_params["n_anchors_target"],
            cell_type_agnostic=cell_type_agnostic,
        )
        adata_out = self._fit_tps_correction(
            adata_pca, self.anchors, regularisation=regularisation
        )
        return adata_out, study, study.best_params

    def integrate_no_optimization(self,
                  adata,
                  nanchors=1000,
                  regularisation=1.0,
                  X_ranked=None,
                  cell_type_agnostic=False,
                  chunk_size=512,
                  min_anchor_weight=0.3):
        """Run anchor selection and TPS correction once, without hyperparameter search.

        The fixed-parameter counterpart to :meth:`optimize`, for when the anchor count
        is already known. Stores the selected anchors on ``self.anchors`` and corrects
        ``adata`` in place.

        :param adata: Data carrying an uncorrected ``obsm["X_pca"]`` plus
            ``obs["species"]`` and ``obs["cell_type"]``. Modified in place.
        :type adata: anndata.AnnData
        :param nanchors: Anchors to select per species pair.
        :type nanchors: int
        :param regularisation: Ridge term on the TPS kernel diagonal.
        :type regularisation: float
        :param X_ranked: Precomputed rank-transformed embedding, or ``None`` to compute it.
        :type X_ranked: numpy.ndarray or None
        :param cell_type_agnostic: If True, match across the whole species pair rather
            than within shared cell types.
        :type cell_type_agnostic: bool
        :param chunk_size: Row block size for the correlation matrix product.
        :type chunk_size: int
        :param min_anchor_weight: Minimum rank correlation for a pair to be retained.
        :type min_anchor_weight: float
        :returns: The corrected data.
        :rtype: anndata.AnnData
        """
        self._precompute_anchor_candidates(
            adata,
            X_ranked=X_ranked,
            cell_type_agnostic=cell_type_agnostic,
            chunk_size=chunk_size,
            min_anchor_weight=min_anchor_weight,
        )
        self.anchors = self._build_anchors(
            n_anchors_target=nanchors,
            cell_type_agnostic=cell_type_agnostic,
        )
        if not self.anchors:
            raise ValueError(
                "No anchors selected — try lowering min_anchor_weight."
            )
        return self._fit_tps_correction(
            adata, self.anchors, regularisation=regularisation
        )

    def post_integration_correction(
            self,
            adata,
            group_cols: list[str],
            method: str = "pca",  # "pca" or "umap"
            pval_cutoff: float = 0.01,
            # UMAP generation params — ignored if method="pca"
            n_neighbors: int = 15,
            min_dist: float = 0.1,
            random_state: int = 42,
    ) -> sc.AnnData:
        """
        Remove outlier cells per group using robust Mahalanobis distance (MCD estimator)
        on either PCA or UMAP coordinates.

        Parameters
        ----------
        adata       : AnnData object output from CSHint_integration
        group_cols  : list of .obs columns to group by e.g. ["species", "ct"]
        method      : "pca" uses .obsm["X_pca"]; "umap" generates a UMAP first
                      then uses .obsm["X_umap"]
        pval_cutoff : chi-squared p-value threshold; cells below this are removed
        n_neighbors : (umap only) number of neighbours for sc.pp.neighbors
        min_dist    : (umap only) min_dist passed to sc.tl.umap
        random_state: (umap only) random seed for reproducibility

        Returns
        -------
        AnnData with outlier cells removed, plus adata.obs["_outlier"] bool column
        """
        adata = adata.copy()

        if method == "umap":
            print("Computing neighbours and UMAP...")
            sc.pp.neighbors(adata, use_rep="X_pca", n_neighbors=n_neighbors)
            sc.tl.umap(adata, min_dist=min_dist, random_state=random_state)
            coords_key = "X_umap"
        elif method == "pca":
            coords_key = "X_pca"
        else:
            raise ValueError(f"method must be 'pca' or 'umap', got '{method}'")

        coords_full = adata.obsm[coords_key]  # (n_cells, n_dims)
        n_dims = coords_full.shape[1]
        obs_df = adata.obs[group_cols].copy()

        #Mahalanobis distance
        is_outlier = pd.Series(False, index=adata.obs_names)

        groups = obs_df.groupby(group_cols, observed=True)
        for group_keys, idx in groups.groups.items():
            coords = coords_full[adata.obs_names.isin(idx)]

            min_samples = max(10, 5 * n_dims)
            if len(idx) < min_samples:
                print(f"Skipping {group_keys}: only {len(idx)} cells (need {min_samples})")
                continue

            try:
                mcd = MinCovDet(random_state=random_state).fit(coords)
                d2 = mcd.mahalanobis(coords)
                pvals = chi2.sf(d2, df=n_dims)
                outlier_mask = pvals < pval_cutoff

                is_outlier[idx[outlier_mask]] = True

            except Exception as e:
                print(f"Skipping {group_keys}: MCD failed — {e}")
                continue

        adata.obs["_outlier"] = is_outlier.values
        n_removed = is_outlier.sum()
        print(f"Removing {n_removed} / {len(adata)} cells ({n_removed / len(adata) * 100:.1f}%)")

        return adata[~is_outlier].copy()
