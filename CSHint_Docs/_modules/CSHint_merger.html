def _compute_og_layers(args):
    """
    Groups genes within an orthogroup into distinct 'layers' (rows).
    Uses Spearman correlation of cell-type means.
    Ensures no layer contains more than one gene from the same species.
    """
    og_name, species_genes, species_cell_type_means = args

    all_genes = []
    gene_to_species = {}
    gene_profiles = []

    for species in sorted(species_genes):
        means = species_cell_type_means[species]
        for g in sorted(species_genes[species]):
            if g in means.columns:
                all_genes.append(g)
                gene_to_species[g] = species
                gene_profiles.append(means[g].values)

    if not all_genes:
        return og_name, {}
    if len(all_genes) == 1:
        return og_name, {all_genes[0]: 0}

    gene_pos = {g: i for i, g in enumerate(all_genes)}
    matrix = np.column_stack(gene_profiles)

    if matrix.shape[0] < 3:
        corr = np.zeros((len(all_genes), len(all_genes)))
    else:
        corr, _ = spearmanr(matrix)
        if len(all_genes) == 2:
            corr = np.array([[1, corr], [corr, 1]])
        elif np.isscalar(corr) or np.isnan(corr).all():
            corr = np.zeros((len(all_genes), len(all_genes)))

    corr = np.asarray(corr, dtype=float)
    corr = np.nan_to_num(corr, nan=-1.0)
    np.fill_diagonal(corr, -np.inf)

    assigned = {}
    remaining_genes = sorted(all_genes)
    z_layer = 0

    while remaining_genes:
        seed_gene = remaining_genes.pop(0)
        current_layer = {seed_gene}
        assigned[seed_gene] = z_layer
        species_in_layer = {gene_to_species[seed_gene]}

        added_in_round = True
        while added_in_round:
            added_in_round = False
            best_candidate = None
            best_score = -np.inf

            for candidate in remaining_genes:
                cand_species = gene_to_species[candidate]
                if cand_species in species_in_layer:
                    continue

                cand_idx = gene_pos[candidate]
                layer_idxs = [gene_pos[g] for g in current_layer]
                mean_corr = float(np.mean([corr[cand_idx, l_idx] for l_idx in layer_idxs]))

                if mean_corr > best_score:
                    best_score = mean_corr
                    best_candidate = candidate

            if best_candidate is not None and best_score > -1.0:
                current_layer.add(best_candidate)
                assigned[best_candidate] = z_layer
                remaining_genes.remove(best_candidate)
                species_in_layer.add(gene_to_species[best_candidate])
                added_in_round = True

        z_layer += 1

    return og_name, assigned

class CSHint_merger:
    """
    Merge per-species scRNA-seq into a joint orthogroup-layer matrix.

    Writes BOTH matrix.mtx (raw integer counts) and matrix_normalised.mtx
    (depth-scaled, optionally log1p).

    def __init__(
        self,
        orthofinder_resultdir,
        scrna_dirs,
        output_dir,
        annotation_obs_col,
        ncores=16,
        counts_layer=None,
        normalise=True,
        scale_factor=1e4,
        log1p=True,
        normalise_by="matrix",
        layer_means_on="normalised",
    ):
        """Initialise a cross-species merge run.

        Validates the enumerated options and sets up the empty containers that
        :meth:`load` and the merge steps populate.

        :param orthofinder_resultdir: Directory holding the OrthoFinder results used for
            orthogroup assignment.
        :type orthofinder_resultdir: str
        :param scrna_dirs: Input scRNA-seq paths keyed by species. The keys define the
            species set for the run.
        :type scrna_dirs: dict[str, str]
        :param output_dir: Destination for the merged matrix and its metadata.
        :type output_dir: str
        :param annotation_obs_col: Column in ``adata.obs`` holding the cell type labels.
        :type annotation_obs_col: str
        :param ncores: Worker processes for the parallelised steps.
        :type ncores: int
        :param counts_layer: Layer holding raw counts, e.g. ``"counts"``. ``None`` uses
            ``adata.X``. Set this whenever ``adata.X`` has been normalised in place —
            otherwise ``matrix.mtx`` will contain normalised values under a raw name, and
            every downstream integer check will pass while the NB offset double-corrects
            for depth.
        :type counts_layer: str or None
        :param normalise: Whether to depth-scale counts during the merge.
        :type normalise: bool
        :param scale_factor: Target sum per cell for depth scaling.
        :type scale_factor: float
        :param log1p: Whether to apply ``log1p`` after scaling.
        :type log1p: bool
        :param normalise_by: Denominator for depth scaling. ``"matrix"`` uses the joint
            matrix column sums, which is what the DE offset uses, so the two stay
            consistent. ``"genome"`` uses each cell's total across all genes in its
            original object, which is the true library size but includes genes dropped
            during orthogroup assignment.
        :type normalise_by: str
        :param layer_means_on: Basis for the cell-type mean profiles that drive layer
            assignment, either ``"normalised"`` or ``"raw"``.
        :type layer_means_on: str
        :raises ValueError: If ``normalise_by`` or ``layer_means_on`` is not one of its
            permitted values.
        """
        self.orthofinder_resultdir = orthofinder_resultdir
        self.scrna_dirs = scrna_dirs
        self.species = list(scrna_dirs.keys())
        self.output_dir = output_dir
        self.ncores = ncores
        self.annotation_obs_col = annotation_obs_col

        self.counts_layer = counts_layer
        self.normalise = normalise
        self.scale_factor = float(scale_factor)
        self.log1p = log1p
        if normalise_by not in ("matrix", "genome"):
            raise ValueError("normalise_by must be 'matrix' or 'genome'.")
        self.normalise_by = normalise_by
        if layer_means_on not in ("normalised", "raw"):
            raise ValueError("layer_means_on must be 'normalised' or 'raw'.")
        self.layer_means_on = layer_means_on

        self.samplesizes = {}
        self.matrices = {}
        self.gene_names = {}
        self.cell_labels = []
        self.annotations = {}
        self.anchor_cell_types = []
        self.cell_type_means = {}
        self.gene_layer_assignments = {}
        self.total_counts = {}
        self.counts_are_integer = {}

        self.row_metadata_list = []

    def run(self):
        """Execute the full merge pipeline end to end.

        Loads the OrthoFinder orthogroups and per-species expression data, computes
        anchor cell-type mean profiles, assigns genes to layers by greedy expression
        similarity, fills the joint matrix, optionally normalises it, and writes the
        result to ``self.output_dir``.

        Normalisation runs only when ``self.normalise`` is set; the parameters governing
        each step are those fixed at construction.

        :returns: ``None``. Output is written to disk and to the instance attributes
            populated along the way.
        :rtype: None
        """
        print("Loading OrthoFinder results...")
        self._get_orthogroups()
        print("Loading scRNA-seq data...")
        self._load_all_scRNA_seq()
        print(f"Computing anchor cell type means ({self.layer_means_on})...")
        self._compute_cell_type_means()
        print("Computing greedy functional homolog layering...")
        self._sort_columns_by_expression_similarity()
        print("Filling final matrix...")
        self._fill_matrix()
        if self.normalise:
            print(f"Normalising (scale_factor={self.scale_factor:g}, log1p={self.log1p}, "
                  f"by={self.normalise_by})...")
            self._normalise_matrix()
        print(f"Saving to {self.output_dir}...")
        self._save()



    def _get_orthogroups(self):
        ow = OrthoFinderWrapper()
        self.orthogroups = ow.read_orthogroups(self.orthofinder_resultdir)

    def _load_all_scRNA_seq(self):
        cell_type_sets = []
        for species, path in self.scrna_dirs.items():
            print(f"  Loading {species}...")
            adata = ad.read_h5ad(path)

            if self.counts_layer is not None:
                if self.counts_layer not in adata.layers:
                    raise KeyError(
                        f"Layer '{self.counts_layer}' not found for {species}. "
                        f"Available layers: {list(adata.layers.keys())}"
                    )
                X = adata.layers[self.counts_layer]
            else:
                X = adata.X

            X = X.tocsc() if sp.issparse(X) else sp.csc_matrix(X)

            data = X.data
            is_int = bool(data.size == 0 or np.allclose(data, np.rint(data)))
            self.counts_are_integer[species] = is_int
            if not is_int:
                src = f"layers['{self.counts_layer}']" if self.counts_layer else "X"
                warnings.warn(
                    f"{species}: {src} holds non-integer values, so it does not look "
                    "like raw counts. matrix.mtx will not be a valid input to an NB "
                    "model. Pass counts_layer= to point at the raw counts.",
                    stacklevel=2,
                )
            if data.size and data.min() < 0:
                warnings.warn(
                    f"{species}: negative values present — this is residual-scaled or "
                    "z-scored data, not counts.",
                    stacklevel=2,
                )

            self.matrices[species] = X
            self.total_counts[species] = np.asarray(X.sum(axis=1)).ravel()
            self.gene_names[species] = np.array(adata.var_names)
            self.samplesizes[species] = adata.n_obs

            if self.annotation_obs_col not in adata.obs.columns:
                raise ValueError(f"Column '{self.annotation_obs_col}' not found for {species}.")

            self.annotations[species] = adata.obs[self.annotation_obs_col]
            self.cell_labels.extend([f"{species}_{c}" for c in adata.obs_names])
            cell_type_sets.append(set(adata.obs[self.annotation_obs_col].unique()))

        self.anchor_cell_types = sorted(set.intersection(*cell_type_sets))
        print(f"  Found {len(self.anchor_cell_types)} anchor cell types across all species")

    def _compute_cell_type_means(self):
        """
        Cell-type mean profiles used for the Spearman layering.

        Defaults to normalised means: the Spearman correlation is taken across
        cell types, so a cell type with systematically deeper cells inflates every
        gene's mean there and drags the correlation structure toward sequencing
        depth rather than expression. Set layer_means_on='raw' to reproduce
        assignments from earlier runs.
        """
        use_norm = self.layer_means_on == "normalised"

        for species in self.species:
            mat = self.matrices[species]
            obs = self.annotations[species]
            genes = self.gene_names[species]
            totals = self.total_counts[species]

            inv = np.zeros_like(totals, dtype=np.float64)
            nz = totals > 0
            inv[nz] = self.scale_factor / totals[nz]

            means = {}
            for cell_type in self.anchor_cell_types:
                mask = (obs == cell_type).values
                if mask.sum() == 0:
                    means[cell_type] = np.zeros(mat.shape[1])
                    continue

                sub = mat[mask, :]
                if use_norm:
                    sub = sp.diags(inv[mask]) @ sub
                    if self.log1p:
                        sub = sub.tocsr()
                        sub.data = np.log1p(sub.data)
                means[cell_type] = np.asarray(sub.mean(axis=0)).flatten()

            self.cell_type_means[species] = pd.DataFrame(means, index=genes).T

    def _sort_columns_by_expression_similarity(self):
        tasks = []
        for og_name, row in self.orthogroups.iterrows():
            species_genes = {}
            for species in self.species:
                if pd.notna(row.get(species)):
                    if isinstance(row[species], str):
                        og_genes = [g.strip() for g in row[species].split(",")]
                    else:
                        og_genes = row[species]

                    valid_genes = [g for g in og_genes if g in self.gene_names[species]]
                    if valid_genes:
                        species_genes[species] = valid_genes

            if species_genes:
                tasks.append((og_name, species_genes, self.cell_type_means))

        results = {}
        with ProcessPoolExecutor(max_workers=self.ncores) as executor:
            futures = {executor.submit(_compute_og_layers, task): task for task in tasks}
            for future in as_completed(futures):
                og_name, assignments = future.result()
                if assignments:
                    results[og_name] = assignments

        self.gene_layer_assignments = {og: results[og] for og in sorted(results)}

    def _fill_matrix(self):
        cell_offset = {}
        offset = 0
        for species in self.species:
            cell_offset[species] = offset
            offset += self.samplesizes[species]

        total_cells = sum(self.samplesizes.values())
        gene_idx_lookup = {s: {g: i for i, g in enumerate(self.gene_names[s])} for s in self.species}

        row_labels = []
        rows, cols, vals = [], [], []
        current_row = 0
        self.row_metadata_list = []

        for og_name, assignments in self.gene_layer_assignments.items():
            layers = defaultdict(list)
            for gene, z in assignments.items():
                layers[z].append(gene)

            for z in sorted(layers.keys()):
                layer_genes = layers[z]
                formatted_genes_for_row = []

                for species in self.species:
                    sp_genes = sorted(g for g in layer_genes if g in gene_idx_lookup[species])

                    if sp_genes:
                        gene = sp_genes[0]
                        formatted_genes_for_row.append(f"{species}:{gene}")

                        g_idx = gene_idx_lookup[species][gene]
                        x_offset = cell_offset[species]

                        sparse_col = self.matrices[species][:, g_idx].tocoo()
                        rows.extend([current_row] * sparse_col.nnz)
                        cols.extend(sparse_col.row + x_offset)
                        vals.extend(sparse_col.data)
                    else:
                        formatted_genes_for_row.append(f"{species}:NA")

                row_label = f"{og_name}_rank{z}"
                row_labels.append(row_label)
                self.row_metadata_list.append({
                    "row_id": row_label,
                    "orthogroup": og_name,
                    "rank": z,
                    "genes": ";".join(formatted_genes_for_row)
                })
                current_row += 1

        vals = np.asarray(vals)

        if vals.size and np.allclose(vals, np.rint(vals)):
            vals = np.rint(vals).astype(np.int32)
            dtype = np.int32
        else:
            vals = vals.astype(np.float32)
            dtype = np.float32
            warnings.warn(
                "Assembled matrix is not integer-valued; writing matrix.mtx as float. "
                "It is not usable as an NB response.",
                stacklevel=2,
            )

        self.final_matrix = sp.csr_matrix(
            (vals, (rows, cols)),
            shape=(current_row, total_cells),
            dtype=dtype
        )
        self.matrix_row_labels = row_labels

        self.matrix_total_counts = np.asarray(self.final_matrix.sum(axis=0)).ravel()
        self.genome_total_counts = np.concatenate([self.total_counts[s] for s in self.species])
        retained = self.matrix_total_counts.sum() / max(self.genome_total_counts.sum(), 1)
        print(f"  {current_row} rows x {total_cells} cells; "
              f"{retained:.1%} of total counts retained in orthogroup layers")

    def _normalise_matrix(self):
        """
        Depth-scale the assembled matrix. Kept as a separate matrix rather than
        replacing the raw one, because the NB GAM needs the counts and the offset,
        not pre-scaled values.
        """
        if self.normalise_by == "matrix":
            denom = self.matrix_total_counts.astype(np.float64)
        else:
            denom = self.genome_total_counts.astype(np.float64)

        n_zero = int((denom == 0).sum())
        if n_zero:
            warnings.warn(
                f"{n_zero} cells have a zero denominator under normalise_by="
                f"'{self.normalise_by}'; their normalised column is left at zero.",
                stacklevel=2,
            )

        inv = np.zeros_like(denom)
        nz = denom > 0
        inv[nz] = self.scale_factor / denom[nz]

        norm = (self.final_matrix.astype(np.float64) @ sp.diags(inv)).tocsr()
        if self.log1p:
            norm.data = np.log1p(norm.data)
        self.normalised_matrix = norm.astype(np.float32)

    def _save(self):
        os.makedirs(self.output_dir, exist_ok=True)
        sio.mmwrite(os.path.join(self.output_dir, "matrix.mtx"), self.final_matrix)

        if self.normalise and getattr(self, "normalised_matrix", None) is not None:
            sio.mmwrite(
                os.path.join(self.output_dir, "matrix_normalised.mtx"),
                self.normalised_matrix,
            )

        col_meta_records = []
        for species in self.species:
            ann = self.annotations[species]
            for bc, cell_type in ann.items():
                col_meta_records.append({
                    "cell_id": f"{species}_{bc}",
                    "species": species,
                    "cell_type": cell_type
                })

        col_df = pd.DataFrame(col_meta_records)

        col_df["total_counts_matrix"] = self.matrix_total_counts
        col_df["total_counts_genome"] = self.genome_total_counts
        col_df["n_features_detected"] = np.diff(self.final_matrix.tocsc().indptr)
        col_df.to_csv(os.path.join(self.output_dir, "metadata.txt"), sep='\t', index=False)

        row_df = pd.DataFrame(self.row_metadata_list)

        row_df.to_csv(os.path.join(self.output_dir, "rowmetadata.csv"), index=False)

        pd.Series(col_df["cell_id"]).to_csv(
            os.path.join(self.output_dir, "barcodes.tsv"), index=False, header=False, sep='\t'
        )
        pd.Series(self.matrix_row_labels).to_csv(
            os.path.join(self.output_dir, "features.tsv"), index=False, header=False, sep='\t'
        )

        manifest = {
            "matrix_raw": "matrix.mtx",
            "matrix_normalised": "matrix_normalised.mtx" if self.normalise else None,
            "counts_layer": self.counts_layer,
            "counts_are_integer": self.counts_are_integer,
            "normalise": self.normalise,
            "scale_factor": self.scale_factor,
            "log1p": self.log1p,
            "normalise_by": self.normalise_by,
            "layer_means_on": self.layer_means_on,
            "species": self.species,
            "n_rows": len(self.matrix_row_labels),
            "n_cells": len(col_meta_records),
            "note": "Use matrix.mtx with offset(log(total_counts_matrix)) for NB GAMs. "
                    "matrix_normalised.mtx is already depth-corrected; do not add an offset.",
        }
        with open(os.path.join(self.output_dir, "normalisation.json"), "w") as fh:
            json.dump(manifest, fh, indent=2)

        print(f"Saved {len(self.matrix_row_labels)} rows and {len(col_meta_records)} cells.")
        print("Metadata generated: metadata.txt, rowmetadata.csv, normalisation.json")
        if self.normalise:
            print("Matrices: matrix.mtx (raw counts), matrix_normalised.mtx (scaled)")
