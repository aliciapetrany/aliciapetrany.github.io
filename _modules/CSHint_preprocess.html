    def preprocess_fasta_headers(self, gtf_fasta_map, out_dir=None):
        """
        Rename FASTA headers from raw accessions to gene symbols.

        Detects RefSeq versus Ensembl annotation format independently for each species,
        including whether the annotation is GTF or GFF3, then builds an accession-to-gene
        index and rewrites the headers against it. For each species this writes a renamed
        ``{species}.faa`` alongside a ``{species}_id_map.csv`` recording the mapping that
        was applied.

        :param gtf_fasta_map: Input paths per species, as
            ``{"species": {"gtf": "path/to/file.gtf|.gff", "fasta": "path/to/file.faa"}, ...}``.
        :type gtf_fasta_map: dict[str, dict[str, str]]
        :param out_dir: Destination for the renamed FASTAs and ID maps. Created if absent.
            Defaults to ``self.output_dir``.
        :type out_dir: str or None
        :returns: ``None``. Output is written to disk.
        :rtype: None
        """

        out_dir = out_dir or self.output_dir
        os.makedirs(out_dir, exist_ok=True)

        for species, paths in gtf_fasta_map.items():
            fmt = self._detect_format(paths["fasta"], paths["gtf"])
            is_gff = self._is_gff(paths["gtf"])
            label = f"{fmt} GFF3" if is_gff else fmt
            print(f"[{species}] Converting FASTA headers... (detected format: {label})")

            id_to_info = self._build_gtf_index(paths["gtf"], species, fmt, is_gff)
            self._detect_link_field(paths["fasta"], id_to_info, species, fmt)

            out_fasta = os.path.join(out_dir, f"{species}.faa")
            out_csv = os.path.join(out_dir, f"{species}_id_map.csv")
            self._convert_fasta(paths["fasta"], id_to_info, out_fasta, out_csv, species, fmt)



    # ── private helpers ───────────────────────────────────────────────────────

    @staticmethod
    def _open(path):
        return gzip.open(path, "rt") if str(path).endswith((".gz", ".bgz")) else open(path)

    @staticmethod
    def _parse_gtf_attrs(attr_string):
        attrs = {}
        for m in re.finditer(r'(\w+)\s+"([^"]*)"', attr_string):
            attrs[m.group(1)] = m.group(2)
        return attrs

    @staticmethod
    def _parse_gff3_attrs(attr_string):
        """GFF3 attributes are 'key=value;key=value', percent-encoded (e.g. %2C -> ,)."""
        attrs = {}
        for field in attr_string.strip().split(";"):
            if not field or "=" not in field:
                continue
            key, val = field.split("=", 1)
            attrs[key.strip()] = unquote(val.strip())
        return attrs

    @staticmethod
    def _is_gff(path):
        """Format is detected from the annotation filename, per the caller's request."""
        p = str(path)
        if p.endswith((".gz", ".bgz")):
            p = p.rsplit(".", 1)[0]
        return p.lower().endswith((".gff", ".gff3"))

    @staticmethod
    def _versioned(base, version):
        """Rebuild an Ensembl-style versioned accession: (ENSGALG00000038044, '1') -> ENSGALG00000038044.1"""
        if not base:
            return ""
        return f"{base}.{version}" if version else base

    def _detect_format(self, fasta_path, gtf_path, sample_n=25):
        """
        Sniff RefSeq vs Ensembl from cheap signals in the first few lines of each file:
          - Ensembl FASTA headers carry inline 'gene:...  transcript:...' tags.
          - Ensembl GTFs carry 'gene_version' / 'transcript_version' attributes,
            which RefSeq GTFs never emit (RefSeq bakes the version into the
            accession itself, e.g. NP_000005.2).
        """
        fasta_hits = 0
        with self._open(fasta_path) as fh:
            seen = 0
            for line in fh:
                if not line.startswith(">"):
                    continue
                seen += 1
                if re.search(r'\bgene:\S+', line) and re.search(r'\btranscript:\S+', line):
                    fasta_hits += 1
                if seen >= sample_n:
                    break

        gtf_hits = 0
        with self._open(gtf_path) as fh:
            seen = 0
            for line in fh:
                if line.startswith("#") or not line.strip():
                    continue
                seen += 1
                if "gene_version" in line or "transcript_version" in line:
                    gtf_hits += 1
                if seen >= sample_n:
                    break

        return "ensembl" if (fasta_hits or gtf_hits) else "refseq"

    # --- FASTA header parsing, per format ---------------------------------------

    _FASTA_PATTERNS_REFSEQ = [
        re.compile(r"^>?\s*((?:NP|XP|YP|WP)_\S+)"),
        re.compile(r"^>?\s*((?:NM|XM|NR|XR)_\S+)"),
        re.compile(r"GeneID:(\d+)"),
        re.compile(r"^>?\s*(\S+)"),
    ]

    _ENSEMBL_KV_PATTERN = re.compile(r'^(\w+):(.+)$')

    def _parse_ensembl_header(self, header):
        """
        Ensembl pep FASTA headers look like:
          >ENSGALP00000049024.1 pep scaffold:... gene:ENSGALG...1 transcript:ENSGALT...1
           gene_biotype:protein_coding transcript_biotype:protein_coding
           gene_symbol:SPATS2 description:...
        Returns (primary_id, {key: value}) pulled from the space-delimited key:value
        tags. Multi-word values (e.g. 'description') only capture their first word,
        which is fine since we never use that field.
        """
        h = header.lstrip(">").strip()
        tokens = h.split()
        primary_id = tokens[0] if tokens else ""
        kv = {}
        for tok in tokens[1:]:
            m = self._ENSEMBL_KV_PATTERN.match(tok)
            if m:
                kv.setdefault(m.group(1), m.group(2))
        return primary_id, kv

    def _get_fasta_candidates(self, header, fmt):
        """
        Ordered list of accessions to try against the GTF index.
        RefSeq: pull NP_/XP_/... style accessions out of free text.
        Ensembl: use the structured primary id plus the gene:/transcript: tags.
        """
        if fmt == "ensembl":
            primary_id, kv = self._parse_ensembl_header(header)
            raw = [primary_id, kv.get("transcript"), kv.get("gene")]
            seen, ordered = set(), []
            for c in raw:
                if c and c not in seen:
                    seen.add(c)
                    ordered.append(c)
            return ordered

        candidates = []
        h = header.lstrip(">").strip()
        for pat in self._FASTA_PATTERNS_REFSEQ:
            m = pat.search(h)
            if m and m.group(1) not in candidates:
                candidates.append(m.group(1).strip())
        return candidates


    def _build_gtf_index(self, gtf_path, species, fmt, is_gff=False):
        if fmt == "ensembl":
            return self._build_gtf_index_ensembl(gtf_path, species)
        if is_gff:
            return self._build_gtf_index_refseq_gff(gtf_path, species)
        return self._build_gtf_index_refseq(gtf_path, species)

    def _build_gtf_index_refseq(self, gtf_path, species):
        id_to_info = {}
        with self._open(gtf_path) as fh:
            for line in fh:
                if line.startswith("#") or not line.strip():
                    continue
                parts = line.rstrip("\n").split("\t")
                if len(parts) < 9:
                    continue

                nc_acc = parts[0]
                attrs = self._parse_gtf_attrs(parts[8])

                gene_id = attrs.get("gene_id", "")
                transcript_id = attrs.get("transcript_id", "")
                gene_symbol = attrs.get("gene", "") or gene_id

                protein_ids = []
                if "protein_id" in attrs:
                    protein_ids.append(attrs["protein_id"])
                for m in re.finditer(r'db_xref\s+"(?:GenBank:)?((?:NP|XP|YP)_\S+?)"', parts[8]):
                    protein_ids.append(m.group(1).rstrip('"'))

                info = dict(gene_id=gene_id, transcript_id=transcript_id,
                            nc_accession=nc_acc, gene_symbol=gene_symbol)

                for acc in filter(None, [gene_id, transcript_id] + protein_ids):
                    id_to_info.setdefault(acc.strip(), info)

        print(f"  [{species}] GTF index (RefSeq): {len(id_to_info):,} accession entries.")
        return id_to_info

    def _build_gtf_index_refseq_gff(self, gtf_path, species):
        """
        RefSeq GFF3 attributes are 'key=value' rather than GTF's 'key "value"',
        and RefSeq repeats 'gene=SYMBOL' onto every child feature (mRNA/exon/CDS),
        so no parent-lookup pass is needed to resolve a symbol. Protein and
        transcript accessions come from explicit 'protein_id='/'transcript_id='
        attributes on CDS/RNA features; the 'Genbank:' entry in Dbxref is used
        as a fallback when those attributes are absent, and 'GeneID:' in Dbxref
        supplies a stable numeric gene id.
        """
        id_to_info = {}
        rna_features = {"mRNA", "lnc_RNA", "ncRNA", "transcript", "misc_RNA",
                        "tRNA", "rRNA", "snRNA", "snoRNA", "primary_transcript"}

        with self._open(gtf_path) as fh:
            for line in fh:
                if line.startswith("#") or not line.strip():
                    continue
                parts = line.rstrip("\n").split("\t")
                if len(parts) < 9:
                    continue

                nc_acc = parts[0]
                feature = parts[2]
                attrs = self._parse_gff3_attrs(parts[8])

                gene_symbol = attrs.get("gene", "")

                gene_ncbi_id, genbank_acc = "", ""
                for entry in attrs.get("Dbxref", "").split(","):
                    if entry.startswith("GeneID:"):
                        gene_ncbi_id = entry.split(":", 1)[1]
                    elif entry.startswith("Genbank:"):
                        genbank_acc = entry.split(":", 1)[1]

                transcript_id = attrs.get("transcript_id", "")
                protein_id = attrs.get("protein_id", "")
                if not transcript_id and feature in rna_features:
                    transcript_id = genbank_acc
                if not protein_id and feature == "CDS":
                    protein_id = genbank_acc

                if not (gene_symbol or transcript_id or protein_id):
                    continue  # region/exon-only lines etc. — nothing worth indexing

                info = dict(gene_id=gene_ncbi_id or gene_symbol, transcript_id=transcript_id,
                            nc_accession=nc_acc, gene_symbol=gene_symbol)

                for acc in filter(None, [gene_symbol, transcript_id, protein_id]):
                    id_to_info.setdefault(acc.strip(), info)

        print(f"  [{species}] GTF index (RefSeq GFF3): {len(id_to_info):,} accession entries.")
        return id_to_info

    def _build_gtf_index_ensembl(self, gtf_path, species):
        """
        Ensembl GTFs split accession and version into separate attributes
        (gene_id/gene_version, transcript_id/transcript_version,
        protein_id/protein_version) and store the gene symbol under 'gene_name'
        rather than 'gene'. Protein ids only appear on CDS lines. We reconstruct
        versioned ids (e.g. ENSGALG00000038044.1) so they match the versioned
        accessions used in Ensembl FASTA headers.
        """
        id_to_info = {}
        with self._open(gtf_path) as fh:
            for line in fh:
                if line.startswith("#") or not line.strip():
                    continue
                parts = line.rstrip("\n").split("\t")
                if len(parts) < 9:
                    continue

                nc_acc = parts[0]
                attrs = self._parse_gtf_attrs(parts[8])

                gene_id = self._versioned(attrs.get("gene_id", ""), attrs.get("gene_version"))
                transcript_id = self._versioned(attrs.get("transcript_id", ""), attrs.get("transcript_version"))
                protein_id = self._versioned(attrs.get("protein_id", ""), attrs.get("protein_version"))
                gene_symbol = attrs.get("gene_name", "") or attrs.get("gene_id", "")

                info = dict(gene_id=gene_id, transcript_id=transcript_id,
                            nc_accession=nc_acc, gene_symbol=gene_symbol)

                for acc in filter(None, [gene_id, transcript_id, protein_id]):
                    id_to_info.setdefault(acc, info)

        print(f"  [{species}] GTF index (Ensembl): {len(id_to_info):,} accession entries.")
        return id_to_info

    def _detect_link_field(self, fasta_path, id_to_info, species, fmt, sample_n=200):
        counts = defaultdict(int)
        total = 0
        with self._open(fasta_path) as fh:
            for line in fh:
                if not line.startswith(">"):
                    continue
                total += 1
                for cand in self._get_fasta_candidates(line, fmt):
                    if cand in id_to_info:
                        prefix = cand.split("_")[0] if "_" in cand else cand.split(".")[0][:6]
                        counts[prefix] += 1
                        break
                if total >= sample_n:
                    break

        if counts:
            best = max(counts, key=counts.__getitem__)
            report = ", ".join(f"{k}:{v}" for k, v in sorted(counts.items()))
            print(f"  [{species}] Autodetected link field '{best}' ({report}) from {total} headers.")
        else:
            print(f"  [{species}] WARNING: No headers matched GTF index in first {total} sampled.")

    def _convert_fasta(self, fasta_path, id_to_info, out_fasta, out_csv, species, fmt):
        renamed, not_found, csv_rows = 0, 0, []

        sequences = {}  # gene_symbol -> {"seq": str, "csv_row": dict}

        with self._open(fasta_path) as fh:
            candidates, info, header_line, buffer = [], None, "", []

            def process():
                orig = candidates[0] if candidates else "UNKNOWN"
                seq = "".join(l.strip() for l in buffer)

                if info:
                    sym = info["gene_symbol"] or info["gene_id"] or orig
                    row = {"original_id": orig, **info}
                elif fmt == "ensembl":

                    _, kv = self._parse_ensembl_header(header_line)
                    fallback_sym = kv.get("gene_symbol")
                    if fallback_sym:
                        sym = fallback_sym
                        row = {"original_id": orig, "gene_symbol": fallback_sym,
                               "gene_id": kv.get("gene", ""), "transcript_id": kv.get("transcript", ""),
                               "nc_accession": ""}
                    else:
                        sym = f"{orig}_NO_MATCH"
                        row = {"original_id": orig, "gene_symbol": "NO_MATCH",
                               "gene_id": "", "transcript_id": "", "nc_accession": ""}
                else:
                    sym = f"{orig}_NO_MATCH"
                    row = {"original_id": orig, "gene_symbol": "NO_MATCH",
                           "gene_id": "", "transcript_id": "", "nc_accession": ""}

                if sym not in sequences or len(seq) > len(sequences[sym]["seq"]):
                    sequences[sym] = {"seq": seq, "csv_row": row}

            for line in fh:
                if line.startswith(">"):
                    if buffer:
                        process()
                    header_line = line
                    candidates = self._get_fasta_candidates(line, fmt)
                    info = next((id_to_info[c] for c in candidates if c in id_to_info), None)
                    buffer = []
                else:
                    buffer.append(line)
            if buffer:
                process()

        with open(out_fasta, "w") as out:
            for sym, entry in sequences.items():
                out.write(f">{sym}\n")
                seq = entry["seq"]
                for i in range(0, len(seq), 60):
                    out.write(seq[i:i + 60] + "\n")
                row = entry["csv_row"]
                if row["gene_symbol"] == "NO_MATCH":
                    not_found += 1
                else:
                    renamed += 1
                csv_rows.append(row)

        with open(out_csv, "w", newline="") as csvfile:
            writer = csv.DictWriter(csvfile, fieldnames=["original_id", "gene_symbol",
                                                         "gene_id", "transcript_id", "nc_accession"])
            writer.writeheader()
            writer.writerows(csv_rows)

        print(f"  [{species}] {renamed + not_found} unique symbols written ({not_found} unmatched).")
        print(f"  [{species}] → {out_fasta}")
        print(f"  [{species}] → {out_csv}")

    def synchronize_scrna_and_proteomes(self, species_map, prot_out_dir=None, scrna_out_dir=None):
        """
         Filter scRNA-seq and protein sequences to the genes shared between them.

        FASTA headers are assumed to already be gene symbols. Both gene names and record
        IDs are upper-cased before intersecting, so the two sources need only agree up to
        case. Each species is written out as a filtered ``.h5ad`` and ``.faa`` pair named
        after the species key; species with no shared genes are skipped with a warning
        rather than raising.

        Note that duplicate gene symbols created by upper-casing are disambiguated with a
        ``-`` suffix, which will not match any FASTA record and so drops those genes from
        the intersection.

        :param species_map: Input paths per species, as
            ``{"species_name": {"h5ad": "path.h5ad", "fasta": "path.faa"}, ...}``.
        :type species_map: dict[str, dict[str, str]]
        :param prot_out_dir: Directory for the filtered proteomes. Created if absent.
        :type prot_out_dir: str
        :param scrna_out_dir: Directory for the filtered expression data. Created if absent.
        :type scrna_out_dir: str
        :returns: ``None``. Output is written to disk.
        :rtype: None
        :raises TypeError: If either output directory is left as ``None``.
        """

        os.makedirs(prot_out_dir, exist_ok=True)
        os.makedirs(scrna_out_dir, exist_ok=True)

        for species, paths in species_map.items():
            print(f"[{species}] Harmonizing scRNA-seq and Proteome...")

            adata = sc.read_h5ad(paths["h5ad"])
            adata.var_names_make_unique()

            adata.var_names = adata.var_names.str.upper()
            adata.var_names_make_unique(join="-")

            protein_seqs = {record.id.upper(): record for record in SeqIO.parse(paths["fasta"], "fasta")}

            for upper_id, record in protein_seqs.items():
                record.id = upper_id
                record.name = upper_id
                record.description = upper_id

            rna_genes = set(adata.var_names)
            prot_genes = set(protein_seqs.keys())
            shared_genes = sorted(list(rna_genes.intersection(prot_genes)))

            print(f"  [{species}] Shared genes found: {len(shared_genes)}")

            if not shared_genes:
                print(f"  [{species}] WARNING: No shared genes. Skipping.")
                continue

            adata_filtered = adata[:, shared_genes].copy()

            filtered_proteins = [protein_seqs[gene] for gene in shared_genes]

            h5ad_out = os.path.join(scrna_out_dir, f"{species}.h5ad")
            faa_out = os.path.join(prot_out_dir, f"{species}.faa")

            adata_filtered.write(h5ad_out)
            with open(faa_out, "w") as f:
                SeqIO.write(filtered_proteins, f, "fasta")

            print(f"  [{species}] Saved filtered h5ad to {h5ad_out}")
            print(f"  [{species}] Saved filtered proteome to {faa_out}")
