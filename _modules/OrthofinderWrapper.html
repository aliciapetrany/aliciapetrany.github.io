    def run_orthofinder(self, inputdir, orthofinder_path, threads = 16):
        """Run OrthoFinder over a directory of proteome FASTA files.

        Invokes the OrthoFinder executable as a shell command and blocks until it
        finishes. Results are written to OrthoFinder's own output directory beneath
        ``inputdir``, not returned; pass that path to :meth:`read_orthogroups` afterwards.

        :param inputdir: Directory of proteome FASTA files to compare.
        :type inputdir: str
        :param orthofinder_path: Path to the OrthoFinder executable.
        :type orthofinder_path: str
        :param threads: Threads for OrthoFinder to use. Also updates ``self.threads``.
        :type threads: int
        :returns: ``None``. Output is written to disk.
        :rtype: None
        """
        if self.threads != threads:
            self.threads = threads

        cmd = [
            orthofinder_path,
            "-f", str(inputdir),
            "-t", str(threads)
        ]
        print("Running:", " ".join(cmd))
        os.system(" ".join(cmd))



    def _genes_from_h5(self, h5_path: str) -> list[str]:
        adata = ad.read_h5ad(h5_path) if h5_path.endswith(".h5ad") else ad.read_10x_h5(h5_path)
        return adata.var_names.tolist()

    def _genes_from_mtx_dir(self, dir_path: str) -> list[str]:

        candidates = [
            "features.tsv.gz", "features.tsv",
            "genes.tsv.gz", "genes.tsv",
        ]

        for filename in candidates:
            filepath = os.path.join(dir_path, filename)
            if not os.path.exists(filepath):
                continue

            open_fn = gzip.open if filename.endswith(".gz") else open

            genes = []
            with open_fn(filepath, "rt") as fh:
                for line in fh:
                    parts = line.rstrip("\n").split("\t")
                    gene = parts[1] if len(parts) > 1 else parts[0]
                    genes.append(gene)

            return genes

        raise ValueError(
            f"No features/genes TSV file found in MTX directory: {dir_path}\n"
            f"Expected one of: {candidates}"
        )

    def _genes_from_csv(self, csv_path: str) -> list[str]:
        gene_col_aliases = {"gene", "gene_name", "genes", "gene_names"}

        with open(csv_path, newline="") as fh:
            reader = csv.DictReader(fh)
            headers = [h.lower().strip() for h in (reader.fieldnames or [])]

            target = next(
                (reader.fieldnames[i]
                 for i, h in enumerate(headers)
                 if h in gene_col_aliases),
                None,
            )

            if target is None and reader.fieldnames:
                target = reader.fieldnames[0]

            if target is None:
                raise ValueError(f"CSV file appears to be empty: {csv_path}")

            return [row[target] for row in reader if row[target].strip()]

    def read_orthogroups(self, orthofinder_dir:str) -> pd.DataFrame:
        """Run OrthoFinder over a directory of proteome FASTA files.

        Invokes the OrthoFinder executable as a shell command and blocks until it
        finishes. Results are written to OrthoFinder's own output directory beneath
        ``inputdir``, not returned; pass that path to :meth:`read_orthogroups` afterwards.

        :param inputdir: Directory of proteome FASTA files to compare.
        :type inputdir: str
        :param orthofinder_path: Path to the OrthoFinder executable.
        :type orthofinder_path: str
        :param threads: Threads for OrthoFinder to use. Also updates ``self.threads``.
        :type threads: int
        :returns: ``None``. Output is written to disk.
        :rtype: None
        """
        tsv_path = os.path.join(orthofinder_dir, "Orthogroups", "Orthogroups.tsv")
        if not os.path.exists(tsv_path):
            raise FileNotFoundError(
                f"Orthogroups TSV not found at expected path: {tsv_path}"
            )
        return pd.read_csv(tsv_path, sep="\t", index_col=0)
